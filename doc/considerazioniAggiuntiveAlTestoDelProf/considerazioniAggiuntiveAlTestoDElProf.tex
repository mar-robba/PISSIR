================================================================
1. DATABASE CENTRALE
================================================================

Il database centrale raccoglie tutte le informazioni provenienti dai
microservizi (stazioni e treni) e le storicizza. Le tabelle si dividono
in tabelle operative (stato corrente) e tabelle storiche (dati
storicizzati).


----------------------------------------------------------------
1.1 Tabelle Operative
----------------------------------------------------------------

--- Stazione ---

CREATE TABLE Stazione (
    id_stazione                  INT PRIMARY KEY,
    nome                         VARCHAR(100) NOT NULL,
    tipoCapolineaPartenzaoNormale VARCHAR(50)
        CHECK (tipoCapolineaPartenzaoNormale
               IN ('capolinea','partenza','normale'))
);

--- Tratte ---

CREATE TABLE Tratte (
    id_Tratta          INT PRIMARY KEY,
    StazionePartenzaFK INT NOT NULL,
    StazioneArrivoFK   INT NOT NULL,
    FOREIGN KEY (StazionePartenzaFK) REFERENCES Stazione(id_stazione),
    FOREIGN KEY (StazioneArrivoFK)   REFERENCES Stazione(id_stazione)
);

--- Utenti ---

CREATE TABLE Utenti (
    id_utente  INT PRIMARY KEY,
    tipo       VARCHAR(50) NOT NULL,   -- es. 'operatore','tecnico','admin'
    nome       VARCHAR(100) NOT NULL,
    cognome    VARCHAR(100) NOT NULL,
    matricola  VARCHAR(50)  UNIQUE NOT NULL
);

--- Treni ---

CREATE TABLE Treni (
    id_convoglio                    INT PRIMARY KEY,
    stato                           VARCHAR(30) NOT NULL
        CHECK (stato IN ('attivo','fermo','rotto','in manutenzione')),
    itinerario                      INT,
    PosizioneAttualeTrattaOStazione INT,
    FOREIGN KEY (itinerario)
        REFERENCES Itinerari(id_itinerario),
    FOREIGN KEY (PosizioneAttualeTrattaOStazione)
        REFERENCES Tratte(id_Tratta)
);

--- Itinerari ---

CREATE TABLE Itinerari (
    id_itinerario INT PRIMARY KEY

);

-- Tabella associativa N:M  Itinerari <-> Tratte
-- Un itinerario e' composto da piu' tratte ordinate
CREATE TABLE Itinerario_Tratta (
    id_itinerario INT NOT NULL,
    id_Tratta     INT NOT NULL,
    ordine        INT NOT NULL,   -- posizione della tratta nell'itinerario
    PRIMARY KEY (id_itinerario, id_Tratta),
    FOREIGN KEY (id_itinerario) REFERENCES Itinerari(id_itinerario),
    FOREIGN KEY (id_Tratta)     REFERENCES Tratte(id_Tratta)
);

--- Transiti ---

-- Relazione (0,N) Stazione  --  (0,1) Transiti
-- "e_spettatrice_correntemente"
CREATE TABLE Transiti (
    id_transito    INT PRIMARY KEY,
    id_stazione    INT NOT NULL,
    id_convoglio   INT NOT NULL,
    id_Tratta      INT,
    tempoEntrata   TIMESTAMP NOT NULL,
    tempoUscita    TIMESTAMP,
    FOREIGN KEY (id_stazione)  REFERENCES Stazione(id_stazione),
    FOREIGN KEY (id_convoglio) REFERENCES Treni(id_convoglio),
    FOREIGN KEY (id_Tratta)    REFERENCES Tratte(id_Tratta)
);

--- Guasti Pervenuti da Treni o Stazioni ---

CREATE TABLE Guasti_Pervenuti_da_treni_o_Staz (
    id_Guasto                    INT PRIMARY KEY,
    Stato_RisoltoONO             BOOLEAN NOT NULL DEFAULT FALSE,
    OperatoreCheSeNeStaOccupandoFK INT,
    FOREIGN KEY (OperatoreCheSeNeStaOccupandoFK)
        REFERENCES Utenti(id_utente)
);

Relazione: perAChiEAssegnataLaRiparazione
La foreign key OperatoreCheSeNeStaOccupandoFK nella tabella
Guasti_Pervenuti_da_treni_o_Staz realizza la relazione con Utenti
("per a chi è assegnata la riparazione").


----------------------------------------------------------------
1.2 Tabelle Storiche (Storici)
----------------------------------------------------------------

Le tabelle storiche conservano una copia immutabile di ogni evento
significativo. Ogni riga possiede un id proprio e i timestamp di
riferimento.

--- Storico Transiti ---

CREATE TABLE Storico_Transiti (
    id_storico_transito INT PRIMARY KEY AUTO_INCREMENT,
    id_transito         INT NOT NULL,
    id_stazione         INT NOT NULL,
    id_convoglio        INT NOT NULL,
    id_Tratta           INT,
    tempoEntrata        TIMESTAMP NOT NULL,
    tempoUscita         TIMESTAMP,
    ts_storicizzazione  TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (id_stazione)  REFERENCES Stazione(id_stazione),
    FOREIGN KEY (id_convoglio) REFERENCES Treni(id_convoglio),
    FOREIGN KEY (id_Tratta)    REFERENCES Tratte(id_Tratta)
);

--- Storico Guasti ---

CREATE TABLE Storico_Guasti (
    id_storico_guasto              INT PRIMARY KEY AUTO_INCREMENT,
    id_Guasto                      INT NOT NULL,
    Stato_RisoltoONO               BOOLEAN NOT NULL,
    OperatoreCheSeNeStaOccupandoFK INT,
    ts_apertura                    TIMESTAMP NOT NULL,
    ts_chiusura                    TIMESTAMP,
    ts_storicizzazione             TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (id_Guasto)
        REFERENCES Guasti_Pervenuti_da_treni_o_Staz(id_Guasto),
    FOREIGN KEY (OperatoreCheSeNeStaOccupandoFK)
        REFERENCES Utenti(id_utente)
);

--- Storico Stato Treni ---

CREATE TABLE Storico_Stato_Treni (
    id_storico_treno                INT PRIMARY KEY AUTO_INCREMENT,
    id_convoglio                    INT NOT NULL,
    stato                           VARCHAR(30) NOT NULL,
    itinerario                      INT,
    PosizioneAttualeTrattaOStazione INT,
    ts_storicizzazione              TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (id_convoglio) REFERENCES Treni(id_convoglio)
);

--- Storico Stato Stazioni ---

CREATE TABLE Storico_Stato_Stazioni (
    id_storico_stazione INT PRIMARY KEY AUTO_INCREMENT,
    id_stazione         INT NOT NULL,
    nome                VARCHAR(100) NOT NULL,
    tipoCapolineaPartenzaoNormale VARCHAR(50),
    funzionanteONo      BOOLEAN,
    ts_storicizzazione  TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (id_stazione) REFERENCES Stazione(id_stazione)
);

--- Storico Itinerari ---

CREATE TABLE Storico_Itinerari (
    id_storico_itinerario INT PRIMARY KEY AUTO_INCREMENT,
    id_itinerario         INT NOT NULL,
    id_convoglio          INT NOT NULL,
    ts_assegnazione       TIMESTAMP NOT NULL,
    ts_completamento      TIMESTAMP,
    ts_storicizzazione    TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (id_itinerario) REFERENCES Itinerari(id_itinerario),
    FOREIGN KEY (id_convoglio)  REFERENCES Treni(id_convoglio)
);

--- Storico Assegnazione Operatori a Guasti ---

CREATE TABLE Storico_Assegnazioni_Guasti (
    id_storico_assegnazione INT PRIMARY KEY AUTO_INCREMENT,
    id_Guasto               INT NOT NULL,
    id_utente               INT NOT NULL,
    ts_assegnazione         TIMESTAMP NOT NULL,
    ts_risoluzione          TIMESTAMP,
    ts_storicizzazione      TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (id_Guasto) REFERENCES
        Guasti_Pervenuti_da_treni_o_Staz(id_Guasto),
    FOREIGN KEY (id_utente) REFERENCES Utenti(id_utente)
);

\begin{itemize}
\item ricezione dai sesori --> modifica della percorrenza della tratta, sincronizzazione sui topic di allarme delle
nuove stazioni
\begin{itemize}
\item se arrivano evendi del tipo la prossima stazione è fuori uso aggiornare variabile in rame e il database
che il treno si è femato e
\item se l'evento è il ripristino della funzionalità della stazione allora riaggiornare la ram e il db che il
treno è partito
\end{itemize}
\end{itemize}
\item Ritardi la tratta possiede orari corretti di percorrenza
\item inviare in certrale l'entrata e uscita da una stazione topic treni/+/passaggio
formato : boh lo fa l'ai
\item invio se il treno si guasta alla centrale
\end{itemize}
